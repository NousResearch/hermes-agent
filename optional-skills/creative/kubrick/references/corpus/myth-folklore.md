  source_family: "Morphology of the Folktale"
  source_records:
    - author: "Vladimir Propp"
      title: "Morphology of the Folktale"
      date: "1928 (English 1958, trans. Laurence Scott)"
      edition: "University of Texas Press / American Folklore Society"
      cultural_context: "Russian formalism applied to 100 Russian fairy tales"
      source_family: "Folklore / Structural Morphology"
      source_tier: PRIMARY
      reliability_context: "Based on specific corpus of Russian tales; highly influential but culture-specific"
      contested_interpretations: "Universality claims vs strict applicability to other traditions"
      access_status: "widely available"
      provenance: "Propp 1928; foundational structural analysis of folktale"
  historical_context: "Tales follow a limited set of 31 functions in fixed sequence. 'Lack' (недостача) initiates the hero's journey; mediation and counteraction follow."
  cultural_scope: ["Russian fairy tales", "influenced narratology globally (Greimas, Lévi-Strauss, Campbell)"]

  observed_structure:
    agents: ["hero", "villain/donor", "lack (missing person/object/quality)"]
    objects: ["the lacked thing", "magical agent or helper"]
    spatial_relation: "home → threshold → other world → return"
    temporal_relation: "initial equilibrium broken by villainy/lack"
    sequence: ["villainy or lack", "mediation (information or call)", "counteraction (hero departs)"]
    transformation: "lack → acquisition of agent → resolution or new lack"
    cost: "loss of prior security or innocence"
    residue: "new status or object carried back"

  candidate_functions:
    narrative: ["standard quest structure", "initiation of dramatic problem"]
    psychological: ["projection of internal lack outward"]
    social: ["community restoration through individual action"]
    ritual: ["rite of passage structure"]
    archetypal: ["hero's journey departure"]

  cinematic_affordances:
    composition: "empty frame or absent character signaling lack"
    blocking: "hero at threshold, looking out or being called"
    camera: "wide establishing shot of home, then closer on departure"
    editing: "ellipsis from lack to call to departure"
    sound: "news of lack delivered by sound or messenger"
    production_design: "home shown as depleted or marked by absence"
    performance: "hero's stillness or hesitation before action"

  mutation_rules:
    inversion: "counteraction fails or creates worse lack"
    fragmentation: "multiple simultaneous lacks"
    transfer: "lack moves from hero to another character"
    absence: "lack is never named directly"
    recurrence: "new lack appears after apparent resolution"
    contamination: "villainy contaminates the helper"
    integration: "acquired agent resolves original lack"

  misuse_risks:
    cliché: "every story forced into Propp functions"
    tradition_flattening: "applying Russian tale morphology to non-folk narratives without adjustment"
    overstatement: "labeling every beat as a Propp function"
    visual_literalism: "literal 'magic object' props without mutation"
    cultural_sensitivity: "universalizing one folk tradition"

  provenance: "PRIMARY: Propp's analysis of Russian tales. Used as structural grammar, not universal template."
  confidence: 0.8
  authority_status: PRIMARY

## Pattern 2: Trickster Crossing and Boundary Violation

**SymbolicNarrativePattern**:
  pattern_id: "trickster_boundary_violation"
  name: "Trickster as Boundary Crosser and Violator"
  source_family: "Comparative Mythology / Folklore"
  source_records:
    - author: "various (e.g., Paul Radin 'The Trickster' 1956; primary tales from Winnebago, Coyote cycles, Loki, Anansi, Hermes)"
      title: "Primary oral and written trickster cycles (e.g., Winnebago Trickster Cycle, Coyote stories)"
      date: "pre-modern oral traditions; recorded 19th-20th c."
      edition: "various ethnographic collections"
      cultural_context: "Multiple: Native American (Winnebago, Navajo), West African (Anansi), Norse (Loki), Greek (Hermes)"
      source_family: "Mythology / Folklore"
      source_tier: PRIMARY (oral traditions recorded ethnographically)
      reliability_context: "Oral transmission; versions vary widely"
      contested_interpretations: "Trickster as culture hero vs disruptor; gender and power readings"
      access_status: "scholarly collections"
      provenance: "Ethnographic records and primary tale collections"
  historical_context: "Trickster figures cross boundaries (human/animal, divine/mortal, life/death), violate norms, and through violation bring change or culture."
  cultural_scope: ["widespread in global mythologies"]

  observed_structure:
    agents: ["trickster", "boundary guardian or norm", "victim or beneficiary"]
    objects: ["disguise", "tool of deception", "stolen or transformed item"]
    spatial_relation: "at or across threshold"
    temporal_relation: "sudden appearance, violation, consequence"
    sequence: ["approach boundary", "violation or trick", "exposure or escape", "residual change"]
    transformation: "norms or identities inverted or hybridized"
    cost: "punishment, loss, or new knowledge for the community"
    residue: "culture, tool, or story left behind"

  candidate_functions:
    narrative: ["catalyst for change", "exposer of hypocrisy", "creator through accident"]
    psychological: ["shadow or unconscious disruptor"]
    social: ["challenger of rigid structures"]
    ritual: ["carnival or inversion rite"]
    archetypal: ["the fool who reveals truth"]

  cinematic_affordances:
    composition: "figure that appears in wrong place or scale"
    blocking: "trickster moves through spaces forbidden to others"
    camera: "sudden reveal or match cut that shows the violation"
    editing: "rapid shifts that mimic the trick"
    sound: "voice or sound that belongs to another identity"
    production_design: "objects that change function under trickster touch"
    performance: "shape-shifting performance, lies told with bodily conviction"

  mutation_rules:
    inversion: "trick backfires on the trickster"
    fragmentation: "trickster splits into multiple forms"
    transfer: "trickster power passes to another character"
    absence: "trickster disappears but effect remains"
    recurrence: "trickster returns in new guise"
    contamination: "boundary violation spreads"
    integration: "violation becomes new norm"

  misuse_risks:
    cliché: "every clever character labeled 'trickster'"
    tradition_flattening: "mixing Coyote with Loki without noting distinct traditions"
    overstatement: "making the entire story about one trick"
    visual_literalism: "obvious disguises or shape changes"
    cultural_sensitivity: "appropriating specific Indigenous trickster figures"

  provenance: "PRIMARY: Ethnographic collections of oral traditions (Radin, etc.). Multiple cultural sources noted separately."
  confidence: 0.75
  authority_status: PRIMARY
