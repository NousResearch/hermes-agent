      access_status: "scholarly editions"

  historical_context: "Signs classified by relation to object: Icon (resemblance), Index (causal connection), Symbol (convention)."
  cultural_scope: ["American philosophy", "influenced film theory (e.g., Deleuze, Eco)"]

  observed_structure:

    objects: ["photograph (icon)", "smoke or footprint (index)", "word or flag (symbol)"]

    temporal_relation: "present resemblance vs past trace vs ongoing convention"

    fragmentation: "icon breaks into multiple indices"
    transfer: "indexical trace moves from object to character"
    absence: "pure symbol with no indexical ground"
    recurrence: "icon returns as index later"
    contamination: "symbolic meaning contaminates an index"
    integration: "icon + index + symbol combine in one shot"

  misuse_risks:
    cliché: "every photograph treated as 'truth' or 'lie'"
    tradition_flattening: "applying Peirce without regard to cultural sign systems"
    overstatement: "classifying every cinematic element into the triad"
    visual_literalism: "on-the-nose use of photographs or footprints"
    cultural_sensitivity: "universalizing triadic model"

  provenance: "PRIMARY: Peirce's papers. Adapted strictly to observable cinematic relations."

  authority_status: PRIMARY
