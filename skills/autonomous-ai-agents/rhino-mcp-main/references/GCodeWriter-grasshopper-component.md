# GCodeWriter - Grasshopper Custom Component Template

**Source:** Extracted from "I Bypassed the Slicer to Print mid Air" workflow  
**Purpose:** Generate G-code directly from Grasshopper geometry

---

## Component Interface

```
Inputs:
  Paths (List[Curve/Polyline])  - Print paths in order
  Settings (JSON/Text)           - Machine/material settings
  OutputPath (String)            - .gcode file path

Outputs:
  GCode (String)                 - Full G-code text
  Stats (JSON)                   - Time, material, bounds
  Preview (Mesh)                 - Visualization mesh
```

---

## GhPython Implementation

```python
"""
GCodeWriter Component for Grasshopper
Generates G-code directly from parametric paths
"""

import json
import Rhino.Geometry as rg
import math

# ─── Input Parameters ───
# x: List[Curve/Polyline] - Print paths in execution order
# settings_json: String - JSON settings (optional, uses defaults if empty)
# output_path: String - Output .gcode file path (optional)

# ─── Default Settings ───
DEFAULTS = {
    "machine": {
        "nozzle_diameter": 0.4,
        "filament_diameter": 1.75,
        "bed_size": [250, 250, 300]
    },
    "material": {
        "type": "PLA",
        "hotend_temp": 200,
        "bed_temp": 60,
        "fan_speed": 255,
        "retraction_distance": 2.0,
        "retraction_speed": 25,
        "extrusion_multiplier": 1.0
    },
    "print": {
        "xy_speed": 1200,      # mm/min
        "z_speed": 300,
        "first_layer_speed": 600,
        "bridge_speed": 600,
        "travel_speed": 9000
    },
    "geometry": {
        "segment_length": 0.8,
        "min_feature_size": 0.4,
        "bridge_angle_threshold": 45
    }
}

# ─── Load Settings ───
if settings_json and settings_json.strip():
    try:
        settings = json.loads(settings_json)
        # Deep merge with defaults
        def deep_merge(base, override):
            for k, v in override.items():
                if k in base and isinstance(base[k], dict) and isinstance(v, dict):
                    deep_merge(base[k], v)
                else:
                    base[k] = v
        deep_merge(DEFAULTS, settings)
        settings = DEFAULTS
    except json.JSONDecodeError:
        settings = DEFAULTS
else:
    settings = DEFAULTS

# ─── Convert Curves to Polylines ───
def curve_to_polyline(curve, seg_len):
    """Convert any curve to polyline with controlled segmentation"""
    if isinstance(curve, rg.PolylineCurve):
        return curve.ToPolyline()
    elif isinstance(curve, rg.Polyline):
        return curve
    else:
        # NURBS, Line, Arc, etc.
        return curve.ToPolyline(seg_len, seg_len, 0.01, 15)

polylines = []
for crv in x:
    if crv:
        poly = curve_to_polyline(crv, settings["geometry"]["segment_length"])
        if poly and poly.Count > 1:
            polylines.append(poly)

# ─── Extrusion Calculation ───
nozzle = settings["machine"]["nozzle_diameter"]
filament = settings["machine"]["filament_diameter"]
mult = settings["material"]["extrusion_multiplier"]

# Cross-sectional areas
nozzle_area = math.pi * (nozzle / 2) ** 2
filament_area = math.pi * (filament / 2) ** 2
area_ratio = nozzle_area / filament_area  # ~0.054 for 0.4/1.75

def calc_extrusion(length):
    """Calculate E-value increment for a segment length"""
    return length * area_ratio * mult

# ─── G-code Generation ───
gcode_lines = []
stats = {
    "total_length": 0.0,
    "total_extrusion": 0.0,
    "print_time_estimate": 0,
    "bounds": {"min": [float('inf')]*3, "max": [float('-inf')]*3},
    "path_count": len(polylines)
}

# Header
gcode_lines.append("; Generated by Grasshopper GCodeWriter")
gcode_lines.append(f"; Nozzle: {nozzle}mm, Filament: {filament}mm")
gcode_lines.append(f"; Paths: {len(polylines)}")

# Startup sequence
gcode_lines.extend([
    f"M104 S{settings['material']['hotend_temp']}",
    f"M140 S{settings['material']['bed_temp']}",
    "G28",
    "G90",
    "M82",
    f"M106 S{settings['material']['fan_speed']}",
    f"M109 S{settings['material']['hotend_temp']}",
    f"M190 S{settings['material']['bed_temp']}",
    # Prime line
    "G1 X10 Y10 Z0.2 F3000",
    "G1 X100 Y10 E15 F1000",
    "G92 E0"
])

current_e = 0.0
retract_dist = settings["material"]["retraction_distance"]
retract_speed = settings["material"]["retraction_speed"] * 60  # mm/s to mm/min
xy_feed = settings["print"]["xy_speed"]
z_feed = settings["print"]["z_speed"]
travel_feed = settings["print"]["travel_speed"]

for i, poly in enumerate(polylines):
    if poly.Count < 2:
        continue
    
    start_pt = poly.Point(0)
    
    # Travel to start (with retraction if not first)
    if i > 0:
        gcode_lines.append(f"G1 E{current_e - retract_dist:.5f} F{retract_speed}")
    
    gcode_lines.append(f"G0 X{start_pt.X:.3f} Y{start_pt.Y:.3f} Z{start_pt.Z:.3f} F{travel_feed}")
    
    # Deretract
    if i > 0:
        gcode_lines.append(f"G1 E{current_e + retract_dist:.5f} F{retract_speed}")
    
    # Print segments
    for j in range(poly.Count - 1):
        p1 = poly.Point(j)
        p2 = poly.Point(j + 1)
        
        seg_len = p1.DistanceTo(p2)
        e_inc = calc_extrusion(seg_len)
        current_e += e_inc
        
        # Determine feedrate based on Z change (bridge detection)
        z_diff = abs(p2.Z - p1.Z)
        if z_diff > 0.05:  # Z move
            feed = z_feed
        elif j > 0 and j < poly.Count - 2:
            # Check if bridge (large XY span with no support)
            # Simplified: use xy_feed, could add bridge detection
            feed = xy_feed
        else:
            feed = xy_feed
        
        gcode_lines.append(
            f"G1 X{p2.X:.3f} Y{p2.Y:.3f} Z{p2.Z:.3f} E{current_e:.5f} F{feed}"
        )
        
        # Update stats
        stats["total_length"] += seg_len
        stats["total_extrusion"] += e_inc
        for axis, val in enumerate([p2.X, p2.Y, p2.Z]):
            stats["bounds"]["min"][axis] = min(stats["bounds"]["min"][axis], val)
            stats["bounds"]["max"][axis] = max(stats["bounds"]["max"][axis], val)

# Shutdown
gcode_lines.extend([
    "G92 E0",
    f"G1 E{-retract_dist:.5f} F{retract_speed}",
    "G1 Z+10 F3000",
    "M104 S0",
    "M140 S0",
    "M84"
])

# ─── Write Output File ───
gcode_text = "\n".join(gcode_lines)
if output_path and output_path.strip():
    try:
        with open(output_path, 'w') as f:
            f.write(gcode_text)
    except Exception as e:
        print(f"Write error: {e}")

# ─── Calculate Time Estimate ───
# Simplified: total_length / avg_speed
avg_speed = (xy_feed + travel_feed) / 2 / 60  # mm/s
stats["print_time_estimate"] = stats["total_length"] / avg_speed if avg_speed > 0 else 0
stats["total_extrusion"] = current_e

# Fix bounds if empty
if stats["bounds"]["min"][0] == float('inf'):
    stats["bounds"] = {"min": [0,0,0], "max": [0,0,0]}

# ─── Create Preview Mesh ───
preview_mesh = rg.Mesh()
for poly in polylines:
    for j in range(poly.Count - 1):
        p1 = poly.Point(j)
        p2 = poly.Point(j + 1)
        # Add as line (renderable)
        preview_mesh.Vertices.Add(p1)
        preview_mesh.Vertices.Add(p2)
        idx = preview_mesh.Vertices.Count - 2
        preview_mesh.Faces.AddFace(idx, idx + 1, idx + 1, idx)  # degenerate quad for line

# ─── Outputs ───
GCode = gcode_text
Stats = json.dumps(stats, indent=2)
Preview = preview_mesh
```

## Usage Example

### Grasshopper Canvas Setup
```
[Surface] → [Populate3D] → [Voronoi3D] 
    ↓
[MeshEdges] → [Pipe (radius=0.6)] 
    ↓
[CurveToPolyline (seg=0.8)] → [GCodeWriter]
    ↓                              ↓
[Preview]                    [GCode file + Stats]
```

### Settings JSON Input
```json
{
  "machine": {"nozzle_diameter": 0.6, "filament_diameter": 1.75},
  "material": {"type": "PETG", "hotend_temp": 235, "bed_temp": 75, "fan_speed": 200},
  "print": {"xy_speed": 1800, "bridge_speed": 800}
}
```

## Installation

1. Copy code into **GhPython** component
2. Set inputs: `x` (List Access, Curve), `settings_json` (Item Access, String), `output_path` (Item Access, String)
3. Set outputs: `GCode`, `Stats`, `Preview`
4. Save as **GCodeWriter.ghuser** for reuse

## Dependencies

- Rhino 7/8 + Grasshopper
- No external plugins required (pure RhinoCommon)
